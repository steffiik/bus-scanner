const STATUS_DAY_KEY = 'busScannerStatusDay'; function resetDailyStatus() { const day = today(); if (localStorage.getItem(STATUS_DAY_KEY) === day) return; localStorage.setItem(STATUS_DAY_KEY, day); localStorage.setItem('busScannerConfirmedCount', '0'); writeArray(HISTORY_KEY, []); // Noch nicht übertragene Scans bleiben erhalten. } function markTransferProblem(item, message) { const q = readArray(QUEUE_KEY); const entry = q.find(x => x.scanId === item.scanId); if (entry) { entry.transferProblem = true; entry.error = message; writeArray(QUEUE_KEY, q); } updateQueueInfo(); }
